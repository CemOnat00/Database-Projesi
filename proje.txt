cmd/server/          → programın başladığı yer (main.go buraya gelecek)

internal/
├── config/          → .env dosyasını okur (DB şifresi, port vs.)
│
├── domain/          → projenin kalbi, saf iş mantığı
│   ├── entity/      → User gibi veritabanı nesneleri
│   ├── dto/         → gelen/giden JSON şekilleri
│   ├── repository/  → "veritabanına şunu yap" sözleşmesi
│   └── service/     → "şu işi yap" sözleşmesi
│
├── infrastructure/  → dış dünyayla bağlantı
│   ├── database/    → PostgreSQL bağlantısı
│   ├── repository/  → veritabanı işlemlerinin gerçek kodu
│   └── service/     → iş mantığının gerçek kodu
│
├── delivery/http/   → internet'ten gelen istekleri karşılar
│   ├── handler/     → "şu adrese istek geldi, ne yapayım?"
│   ├── middleware/  → "token doğru mu?" kontrolü
│   └── router/      → hangi adres hangi handler'a gider
│
└── pkg/             → her yerden kullanılan yardımcı araçlar
    ├── apperror/    → hata yönetimi
    ├── response/    → JSON cevap formatı
    ├── jwt/         → token üretme/doğrulama
    └── password/    → şifre hashleme
